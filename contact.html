/** * 왕가부동산 상담 신청 폼 → 구글 시트 자동 저장 스크립트 * * [설치 방법] * 1. 구글 스프레드시트를 새로 만듭니다. (예: "왕가부동산_상담신청") * 2. 상단 메뉴 [확장 프로그램] > [Apps Script] 클릭 * 3. 기본 코드(Code.gs)를 모두 지우고 아래 코드를 붙여넣기 * 4. 상단 [배포] > [새 배포] 클릭 * - 유형 선택: "웹 앱" * - 설명: 아무거나 입력 * - 실행 계정: 나 * - 액세스 권한: 전체 허용(모든 사용자) ← 반드시 이렇게 설정해야 폼에서 접근 가능 * 5. 배포 후 나오는 "웹 앱 URL"을 복사 * 6. contact.html 파일 안의 SCRIPT_URL = "PASTE_YOUR_APPS_SCRIPT_WEB_APP_URL_HERE" * 부분을 방금 복사한 URL로 교체 * * 참고: 코드를 수정한 뒤에는 [배포] > [배포 관리] > 연필 아이콘 > "새 버전"으로 * 다시 배포해야 수정 사항이 반영됩니다. */ function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 첫 실행 시 헤더가 없으면 자동 생성 if (sheet.getLastRow() === 0) { sheet.appendRow(['접수일시', '이름', '연락처', '보유 오피스텔', '보유 아파트', '문의 내용']); } const data = JSON.parse(e.postData.contents); const newRow = sheet.getLastRow() + 1; sheet.appendRow([ data.submittedAt || new Date().toLocaleString('ko-KR', {timeZone: 'Asia/Seoul'}), data.name || '', '', // 연락처는 아래에서 텍스트 서식으로 별도 입력 data.officetel || '', data.apartment || '', data.message || '' ]); // 연락처(C열)는 반드시 "일반 텍스트" 서식으로 지정한 뒤 값을 넣어야 // 010-1234-5678의 앞자리 0이 사라지지 않습니다. const phoneCell = sheet.getRange(newRow, 3); phoneCell.setNumberFormat('@'); phoneCell.setValue(data.phone || ''); return ContentService.createTextOutput(JSON.stringify({result: 'success'})) .setMimeType(ContentService.MimeType.JSON); }
